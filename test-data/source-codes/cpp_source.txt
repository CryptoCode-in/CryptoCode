#include <iostream>

int main() {
    int a, b, c;
    std::cout << "Enter first number: ";
    std::cin >> a;
    std::cout << "Enter second number: ";
    std::cin >> b;
    std::cout << "Sum = " << (a + b) << std::endl;
    std::cout << "Enter third number: ";
    std::cin >> c;
    std::cout << "Final result = " << (a + b + c) << std::endl;
    return 0;
}
